    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const branchLocationData = @json($branchLocationData);
            const statusBox = document.getElementById('delivery-status-message');

            function updateStatusUI(icon, title, message, colorType) {
                const colors = {
                    green: { bg: 'bg-emerald-100', border: 'border-emerald-500', text: 'text-emerald-600' },
                    yellow: { bg: 'bg-amber-100', border: 'border-amber-500', text: 'text-amber-600' },
                    red: { bg: 'bg-red-100', border: 'border-red-500', text: 'text-brand-red' }
                };
                
                const c = colors[colorType] || colors.yellow;

                statusBox.className = `${c.bg} rounded-[2rem] p-6 mb-16 shadow-xl flex flex-col md:flex-row items-center justify-between gap-4 border-l-8 ${c.border} transition-all duration-500 max-w-4xl mx-auto`;
                statusBox.innerHTML = `
                    <div class="flex items-center gap-4 text-center md:text-left">
                        <div class="w-12 h-12 bg-white rounded-2xl flex items-center justify-center shadow-sm">
                            <i class="fas ${icon} ${c.text} text-xl"></i>
                        </div>
                        <div>
                            <p class="text-[10px] font-black uppercase tracking-widest ${c.text} opacity-70">${title}</p>
                            <p class="text-sm font-bold text-gray-900 uppercase italic leading-none mt-1">${message}</p>
                        </div>
                    </div>
                `;
            }

            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    (position) => {
                        if (branchLocationData) {
                            updateStatusUI('fa-check-circle', 'Lokasi Terdeteksi', `Outlet terdekat: ${branchLocationData.name}. Anda siap memesan!`, 'green');
                        } else {
                            updateStatusUI('fa-info-circle', 'Lokasi Ditemukan', 'Silakan pilih cabang untuk melihat ketersediaan menu.', 'yellow');
                        }
                    },
                    (error) => {
                        updateStatusUI('fa-exclamation-triangle', 'GPS Tidak Aktif', 'Aktifkan GPS untuk mendapatkan estimasi ongkir & outlet terdekat.', 'red');
                    }
                );
            }
        });
    </script>
