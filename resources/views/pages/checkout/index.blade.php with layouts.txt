        }

            if (el.triggerContainer && !el.triggerContainer.contains(e.target)) {
                el.branchOptions.classList.add('hidden');
                el.branchTrigger.classList.remove('border-gray-900');
            }
        });
        window.selectBranchCustom = function(id, name, address) {
            el.branchNameDisplay.innerText = name;
            el.branchTrigger.classList.add('bg-white', 'border-gray-900');
            el.branchTrigger.classList.remove('bg-slate-50', 'border-transparent');
            el.branchOptions.classList.add('hidden');
            el.realSelect.value = id;
            el.realSelect.dispatchEvent(new Event('change'));
        };

        // --- CORE LOGIC ---
        el.realSelect.addEventListener('change', function() {
            const selected = this.options[this.selectedIndex];
            const name = selected.text.trim();
            const address = selected.getAttribute('data-address') || '';
            let mapUrl = selected.getAttribute('data-map-url');
            el.displayPickupName.innerText = name;
            el.displayPickupAddr.innerText = address;
            if (!mapUrl || mapUrl === 'null') {
                mapUrl = `http://maps.google.com/?q=${encodeURIComponent(name + ' ' + address)}`;
            }
            el.btnPickupMap.href = mapUrl;
            calculateTotal();
        });

        const toggleOrderType = () => {
            if(el.orderTypeDelivery.checked) {
                el.secAddr.style.display = 'block';
                el.secPickup.style.display = 'none';
                el.rowFee.classList.remove('hidden');
                el.methodCod.classList.remove('hidden');
                el.methodPickup.classList.add('hidden');
                el.methodCod.querySelector('input').checked = true;
                calculateTotal(); 
            } else {
                el.secAddr.style.display = 'none';
                el.secPickup.style.display = 'block';
                el.rowFee.classList.add('hidden');
                el.methodCod.classList.add('hidden');
                el.methodPickup.classList.remove('hidden');
                el.methodPickup.querySelector('input').checked = true;
                currentFee = 0; 
                updateSummary();
            }
        };
        el.orderTypeDelivery.addEventListener('change', toggleOrderType);
        el.orderTypePickup.addEventListener('change', toggleOrderType);

        // --- CALCULATE ---
        function calculateTotal() {
            if(!el.orderTypeDelivery.checked) {
                currentFee = 0;
                updateSummary();
                return;
            }
            const locId = el.realSelect.value;
            const userLat = el.inpLat.value;
            const userLng = el.inpLng.value;
            if(!locId || !userLat || !userLng) return;

            el.btn.disabled = true;
            el.btn.innerHTML = '<i class="fas fa-spinner animate-spin"></i> HITUNG ONGKIR...';

            fetch('/api/check-delivery', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'X-CSRF-TOKEN': '{{ csrf_token() }}' },
                body: JSON.stringify({ location_id: locId, latitude: userLat, longitude: userLng })
            })
            .then(res => res.json())
            .then(data => {
                if(data.allowed) {
                    currentFee = parseFloat(data.delivery_fee) || 0;
                    el.btn.disabled = false;
                    el.btn.innerHTML = '<span>KONFIRMASI PESANAN</span><i class="fas fa-arrow-right"></i>';
                } else {
                    Swal.fire({
                        icon: 'error', title: 'Kejauhan!', text: data.message,
                        confirmButtonText: 'Ganti ke Pickup', confirmButtonColor: '#ED1C24'
                    }).then((result) => { if (result.isConfirmed) el.orderTypePickup.click(); });
                    currentFee = 0;
                    el.btn.disabled = true;
                }
                updateSummary();
            })
            .catch(err => {
                console.error(err);
                el.btn.disabled = false;
                el.btn.innerText = "ERROR KONEKSI";
            });
        }

        // --- EXTRAS ---
        window.openAddressBook = function() { document.getElementById('modalAddressBook').classList.remove('hidden'); };
        window.selectAddress = function(addr) {
            el.lblDisplay.innerText = addr.label;
            el.addrDisplay.innerText = addr.address;
            el.phoneDisplay.innerText = addr.phone;
            el.inpLat.value = addr.latitude;
            el.inpLng.value = addr.longitude;
            el.inpAddr.value = addr.address;
            document.getElementById('modalAddressBook').classList.add('hidden');
            calculateTotal();
        };

        el.btnPromo.addEventListener('click', function() {
            const code = el.promoCode.value;
            if(!code) return Swal.fire('Eits!', 'Masukkan kode promo dulu.', 'warning');
            const originalText = el.btnPromo.innerHTML;
            el.btnPromo.innerHTML = '<i class="fas fa-spinner animate-spin"></i>';
            fetch('/api/validate-promo', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'X-CSRF-TOKEN': '{{ csrf_token() }}' },
                body: JSON.stringify({ promo_code: code, subtotal: subtotal })
            })
            .then(res => res.json())
            .then(data => {
                el.btnPromo.innerHTML = originalText;
                if(data.success) {
                    currentDiscount = parseFloat(data.discount_amount) || 0;
                    el.rowDisc.classList.remove('hidden');
                    Swal.fire({ icon: 'success', title: 'Hore!', text: data.message, timer: 2000, showConfirmButton: false });
                } else {
                    currentDiscount = 0;
                    el.rowDisc.classList.add('hidden');
                    Swal.fire({ icon: 'error', title: 'Gagal', text: data.message });
                }
                updateSummary();
            })
            .catch(() => { el.btnPromo.innerHTML = originalText; Swal.fire('Error', 'Gagal cek promo.', 'error'); });
        });

        function updateSummary() {
            const numSubtotal = parseFloat(subtotal) || 0;
            const numFee      = parseFloat(currentFee) || 0;
            const numDiscount = parseFloat(currentDiscount) || 0;
            const total = (numSubtotal + numFee) - numDiscount;
            const fmt = (num) => 'Rp ' + num.toLocaleString('id-ID');
            el.fee.innerText = fmt(numFee);
            el.txtDisc.innerText = '- ' + fmt(numDiscount);
            el.total.innerText = fmt(total);
            if(el.totalMobile) el.totalMobile.innerText = fmt(total); // UPDATE MOBILE JUGA
            el.inpFee.value = numFee;
            el.inpDiscount.value = numDiscount;
            el.inpTotal.value = total;
        }

        if(el.inpLat.value && el.realSelect.value) {
            el.realSelect.dispatchEvent(new Event('change'));
        }
    });
    </script>
